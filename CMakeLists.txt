cmake_minimum_required(VERSION 3.22)
project(PongDynamics VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PONG_PLUGIN "Build VST3 and standalone" ON)
add_executable(pong_tests Tests/tests.cpp)
target_include_directories(pong_tests PRIVATE Source)
enable_testing()
add_test(NAME DSP COMMAND pong_tests)
add_executable(pong_features Tests/features.cpp)
target_include_directories(pong_features PRIVATE Source)
add_test(NAME Features COMMAND pong_features)
add_executable(pong_render Tests/render.cpp)
target_include_directories(pong_render PRIVATE Source)
if(PONG_PLUGIN)
 set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE" CACHE PATH "JUCE source")
 if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/JUCE-8.0.12-source.zip")
   message(FATAL_ERROR "JUCE source missing: restore JUCE-8.0.12-source.zip or set JUCE_DIR")
  endif()
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/JUCE-8.0.12-source.zip"
       DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/bundled-framework")
  set(JUCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/bundled-framework/JUCE")
 endif()
 add_subdirectory("${JUCE_DIR}" juce)
 juce_add_plugin(PongDynamics COMPANY_NAME "PADDO & SOUL22" BUNDLE_ID "audio.paddo.pongdynamics"
 PLUGIN_MANUFACTURER_CODE PdSo PLUGIN_CODE PgDy FORMATS VST3 Standalone PRODUCT_NAME "PONG DYNAMICS"
 IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics)
 target_sources(PongDynamics PRIVATE Source/Plugin.cpp)
 target_compile_definitions(PongDynamics PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(PongDynamics PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 add_executable(pong_host_tests Tests/host.cpp)
 target_compile_definitions(pong_host_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(pong_host_tests PRIVATE PongDynamics juce::juce_audio_utils)
 add_test(NAME Processor COMMAND pong_host_tests)
endif()
